## Работа с параметром точности
Написать функцию деления двух чисел с параметром точности epsilon. Точность можно задать вручную или взять из файла settings.ini. Проверить работу тестами.

Код (main.py)
```python
import configparser

def calculate(first, second, epsilon=0.0001):
    if second == 0:
        raise ZeroDivisionError("Деление на ноль")
    result = first / second
    return round(result, 10)


def load_params(file_name="settings.ini"):
    config = configparser.ConfigParser()
    config.read(file_name)
    value = float(config["settings"]["epsilon"])
    return value
```

```python
[settings]
epsilon = 0.0001
```

Тесты (test.py)
```python
import os
import unittest
from main import calculate, load_params

class CalculateTests(unittest.TestCase):
    def test_1(self):
        self.assertEqual(calculate(1, 2, epsilon=0.1), 0.5)
    def test_2(self):
        self.assertEqual(calculate(1, 1000, epsilon=0.001), 0.001)
    def test_3(self):
        with self.assertRaises(ZeroDivisionError):
            calculate(5, 0)

class SettingsTests(unittest.TestCase):
    def test_1(self):
        self.assertTrue(os.path.exists("settings.ini"))
    def test_2(self):
        value = load_params()
        self.assertTrue(10**-9 <= value <= 10**-1)
    def test_3(self):
        value = load_params()
        self.assertIsInstance(value, float)

if __name__ == "__main__":
    unittest.main()
```
